cmake_minimum_required(VERSION 3.20)
project(mini-compiler-mlir LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard to conform to")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

enable_testing()

find_package(LLVM REQUIRED CONFIG)
find_package(MLIR REQUIRED CONFIG)
find_package(CUDAToolkit QUIET)

message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION}")
message(STATUS "LLVM CMake path: ${LLVM_CMAKE_DIR}")
message(STATUS "MLIR CMake path: ${MLIR_CMAKE_DIR}")
if(CUDAToolkit_FOUND)
  message(STATUS "Found CUDA toolkit: ${CUDAToolkit_VERSION}")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17 CACHE STRING "CUDA standard to conform to")
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(MINI_CUDA_ARCHITECTURES "86" CACHE STRING
    "CUDA architectures used by mini compiler CUDA benchmark targets")
  set(MINI_NVTX_ENABLED OFF)
  set(MINI_NVTX_INCLUDE_DIR "")
  set(MINI_NVTX_LINK_LIBS "")
  get_filename_component(MINI_CUDA_ROOT "${CUDAToolkit_BIN_DIR}/.." ABSOLUTE)
  if(EXISTS "${MINI_CUDA_ROOT}/include/nvtx3/nvToolsExt.h")
    set(MINI_NVTX_INCLUDE_DIR "${MINI_CUDA_ROOT}/include")
  elseif(EXISTS "${MINI_CUDA_ROOT}/targets/x86_64-linux/include/nvtx3/nvToolsExt.h")
    set(MINI_NVTX_INCLUDE_DIR "${MINI_CUDA_ROOT}/targets/x86_64-linux/include")
  endif()
  if(TARGET CUDA::nvToolsExt)
    set(MINI_NVTX_ENABLED ON)
    set(MINI_NVTX_LINK_LIBS CUDA::nvToolsExt)
  elseif(CUDA_nvToolsExt_LIBRARY)
    set(MINI_NVTX_ENABLED ON)
    set(MINI_NVTX_LINK_LIBS "${CUDA_nvToolsExt_LIBRARY}")
  elseif(MINI_NVTX_INCLUDE_DIR)
    set(MINI_NVTX_ENABLED ON)
    set(MINI_NVTX_LINK_LIBS ${CMAKE_DL_LIBS})
  endif()
  if(MINI_NVTX_ENABLED)
    if(MINI_NVTX_LINK_LIBS STREQUAL "CUDA::nvToolsExt"
       OR MINI_NVTX_LINK_LIBS STREQUAL "${CUDA_nvToolsExt_LIBRARY}")
      message(STATUS "Found NVTX library support: ${MINI_NVTX_LINK_LIBS}")
    else()
      message(STATUS
        "Found NVTX3 headers without nvToolsExt library; enabling header-only NVTX support")
    endif()
  else()
    message(WARNING "NVTX library not found; NVTX ranges will be compiled out")
  endif()
else()
  message(WARNING "CUDA toolkit not found; building without MiniCudaRuntimeWrappers")
endif()

list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}")

include(TableGen)
include(AddLLVM)
include(AddMLIR)

include_directories(${LLVM_INCLUDE_DIRS})
include_directories(${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

add_definitions(${LLVM_DEFINITIONS})

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

add_subdirectory(lib)
if(CUDAToolkit_FOUND)
  add_subdirectory(runtime)
endif()
add_subdirectory(tools)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/CMakeLists.txt")
  add_subdirectory(test)
endif()
